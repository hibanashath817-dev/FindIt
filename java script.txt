<script>
let reports = JSON.parse(
    localStorage.getItem("finditReports")||"[]"

function showForm(type) {
    document.getElementById("reportSection").style.display = "block";
    document.getElementById("formTitle").innerText =
        "Report " + type + " Item";
    document.getElementById("itemType").value = type;
    document.getElementById("reportSection").scrollIntoView();
}

function closeForm() {
    document.getElementById("reportSection").style.display = "none";
}

document.getElementById("itemForm").addEventListener("submit", function(e) {
    e.preventDefault();

    const report = {
        name: document.getElementById("itemName").value,
        location: document.getElementById("itemLocation").value,
        description: document.getElementById("itemDescription").value,
        type: document.getElementById("itemType").value
    };

    reports.push(report);

localStorage.setItem(
    "finditReports",
    JSON.stringify(reports)
);

displayReports();

    this.reset();
    closeForm();
    alert("Your report was added successfully!");
});

function displayReports() {
    const list = document.getElementById("itemList");
    list.innerHTML = "";

    reports.forEach(item => {
        const card = document.createElement("div");
        card.className = "item-card";

        const heading = document.createElement("h3");
        heading.textContent = item.name;

        const location = document.createElement("p");
        location.textContent = "Location: " + item.location;

        const description = document.createElement("p");
        description.textContent = item.description;

        const status = document.createElement("span");
        status.className = "status";
        status.textContent = item.type;

        card.append(heading, location, description, status);
        list.appendChild(card);
    });
}

function searchItems() {
    const query = document.getElementById("searchInput")
        .value.trim().toLowerCase();

    const list = document.getElementById("itemList");
    list.innerHTML = "";

    const demoItems = [
        {name:"Mobile Phone", location:"College Library",
        description:"Black smartphone", type:"Lost"},
        {name:"Backpack", location:"Classroom",
        description:"Blue school backpack", type:"Found"},
        {name:"Keys", location:"College Garden",
        description:"Set of keys with a keychain", type:"Lost"}
    ];

    const allItems = [...demoItems, ...reports];

    const results = allItems.filter(item =>
        (item.name + " " + item.location + " " +
        item.description + " " + item.type)
        .toLowerCase().includes(query)
    );

    results.forEach(item => {
        const card = document.createElement("div");
        card.className = "item-card";

        const heading = document.createElement("h3");
        heading.textContent = item.name;

        const location = document.createElement("p");
        location.textContent = "Location: " + item.location;

        const description = document.createElement("p");
        description.textContent = item.description;

        const status = document.createElement("span");
        status.className = "status";
        status.textContent = item.type;

        card.append(heading, location, description, status);
        list.appendChild(card);
    });

    if (results.length === 0) {
        list.textContent = "No matching items found.";
    }

    document.getElementById("items").scrollIntoView();
}
window.addEventListener("load", function() {
    if (reports.length > 0) {
        displayReports();
    }
});
